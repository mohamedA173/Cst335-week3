use student_success

// clear it out first in case I already ran this before
db.students.drop()

// validation rules so bad data gets rejected
db.createCollection("students", {
  validator: {
    $jsonSchema: {
      bsonType: "object",
      required: ["studentId", "name", "email", "major", "advisingNotes", "supportVisits"],
      properties: {
        studentId: { bsonType: "string" },
        name: { bsonType: "string" },
        email: { bsonType: "string" },
        major: { bsonType: "string" },
        advisingNotes: {
          bsonType: "array",
          items: {
            bsonType: "object",
            required: ["date", "advisor", "category", "text"],
            properties: {
              date: { bsonType: "date" },
              advisor: { bsonType: "string" },
              category: { enum: ["academic plan", "well-being", "career"] },
              text: { bsonType: "string" }
            }
          }
        },
        supportVisits: {
          bsonType: "array",
          items: {
            bsonType: "object",
            required: ["date", "serviceType"],
            properties: {
              date: { bsonType: "date" },
              serviceType: { enum: ["tutoring", "writing center", "workshop"] },
              description: { bsonType: "string" }
            }
          }
        }
      }
    }
  }
})

// makes sure there are no duplicate students or emails
db.students.createIndex({ studentId: 1 }, { unique: true })
db.students.createIndex({ email: 1 }, { unique: true })
